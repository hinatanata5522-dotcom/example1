function checkShibauraIR() { const url = "https://www.shibaura-machine.co.jp/jp/ir/"; const html = UrlFetchApp.fetch(url).getContentText("UTF-8"); // IRニュース部分だけ抽出 const match = html.match(/IRニュース[\s\S]*?プレスリリース/); if (!match) { Logger.log("IRニュース欄取得失敗"); return; } const currentIR = match[0]; const props = PropertiesService.getScriptProperties(); const oldIR = props.getProperty("shibaura_ir"); // 初回保存 if (!oldIR) { props.setProperty("shibaura_ir", currentIR); Logger.log("初回データ保存"); return; } // 更新検知 if (oldIR !== currentIR) { GmailApp.sendEmail( "ua148768@gmail.com", "【IR更新通知】芝浦機械", "芝浦機械のIRニュースが更新されました。\n\n" + url ); props.setProperty("shibaura_ir", currentIR); Logger.log("IR更新検知"); } }
